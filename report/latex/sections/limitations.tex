% 10. Limitations and Constraints
% Included from main.tex via \input{sections/limitations}
% ---------------------------------------------------------------

\begin{itemize}
    \item \textbf{Location precision.} Locations are known only by five-digit
    zip prefix, so distances carry an error of roughly 1--4\,km at each end and
    are straight-line rather than road distances.
    \item \textbf{New sellers.} 13.4\% of test orders come from sellers not
    seen in training, for whom seller-history features fall back to averages.
    \item \textbf{Short test period.} The test set covers a single month
    (August 2018), and the validation month has an unusually low late rate.
    \item \todo{Further limitations, e.g.\ events such as the May 2018 truckers'
    strike, delivered orders only.}
\end{itemize}
